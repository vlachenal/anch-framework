142 template<typename F>
143 struct Signature {
145 using Args = typename FunctionTraits<F>::Args;
146
148 using Result = std::remove_cvref_t<typename FunctionTraits<F>::Result>;
149
151 static constexpr std::size_t COUNT = std::tuple_size_v<Args>;
152
154 static constexpr bool HAS_REQUEST =
155 std::is_same_v<std::remove_cvref_t<typename ElementOr<0, Args>::Type>, anch::rest::Request>;
156
158 using Last = std::remove_cvref_t<typename ElementOr<COUNT - 1, Args>::Type>;
159
161 static constexpr bool HAS_BODY = COUNT > 0 && IsBody<Last>::value;
162
164 using BodyType = typename IsBody<Last>::Type;
165
167 static constexpr std::size_t PATH_COUNT = COUNT - (HAS_REQUEST ? 1 : 0) - (HAS_BODY ? 1 : 0);
168
174 template<std::size_t I>
175 using PathArg = std::remove_cvref_t<std::tuple_element_t<I + (HAS_REQUEST ? 1 : 0), Args>>;
176
178 static constexpr bool SERIALIZED = !std::is_void_v<Result> && !std::is_same_v<Result, anch::rest::Response>;
179
181 static constexpr bool OPTIONAL = IsOptional<Result>::value;
182
184 using Value = std::conditional_t<SERIALIZED, typename IsOptional<Result>::Type, std::monostate>;
185 };
