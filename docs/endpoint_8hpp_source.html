145 template<typename F>
146 struct Signature {
148 using Args = typename FunctionTraits<F>::Args;
149
151 using Result = std::remove_cvref_t<typename FunctionTraits<F>::Result>;
152
154 static constexpr std::size_t COUNT = std::tuple_size_v<Args>;
155
157 static constexpr bool HAS_REQUEST =
158 std::is_same_v<std::remove_cvref_t<typename ElementOr<0, Args>::Type>, anch::rest::Request>;
159
161 using Last = std::remove_cvref_t<typename ElementOr<COUNT - 1, Args>::Type>;
162
164 static constexpr bool HAS_BODY = COUNT > 0 && IsBody<Last>::value;
165
167 using BodyType = typename IsBody<Last>::Type;
168
170 static constexpr bool RAW_BODY = COUNT > 0 && std::is_same_v<typename ElementOr<COUNT - 1, Args>::Type, std::istream&>;
171
173 static constexpr std::size_t PATH_COUNT = COUNT - (HAS_REQUEST ? 1 : 0) - ((HAS_BODY || RAW_BODY) ? 1 : 0);
174
180 template<std::size_t I>
181 using PathArg = std::remove_cvref_t<std::tuple_element_t<I + (HAS_REQUEST ? 1 : 0), Args>>;
182
184 static constexpr bool SERIALIZED = !std::is_void_v<Result> && !std::is_same_v<Result, anch::rest::Response>;
185
187 static constexpr bool OPTIONAL = IsOptional<Result>::value;
188
190 using Value = std::conditional_t<SERIALIZED, typename IsOptional<Result>::Type, std::monostate>;
191 };
