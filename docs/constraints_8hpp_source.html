245 if constexpr (anch::json::MapLike<V>) {
246 return (IsValueConstraint<C, Unwrap<typename V::mapped_type>>::value && ...);

248 return (IsValueConstraint<C, Unwrap<std::ranges::range_value_t<V>>>::value && ...);
249 } else {
