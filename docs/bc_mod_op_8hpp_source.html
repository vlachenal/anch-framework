337 inline void
338 BlockCipherModeOfOperation<Derived,Cipher>::cipherSequentially(std::istream& input,
339 std::ostream& output,
340 const std::string& key) {
341 Cipher cipher(reinterpret_cast<const uint8_t*>(key.data()));
342 std::array<uint8_t, Cipher::getBlockSize()> data;
343 std::array<uint8_t, Cipher::getBlockSize()> out;
344 uint32_t index = 0;
345 while(!input.eof()) {
346 input.read(reinterpret_cast<char*>(data.data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
347 std::streamsize nbRead = input.gcount();
348 if(nbRead == 0) {
349 break;
350 }
351 std::size_t nbBlocks = cipherBlock(data, nbRead, out, index, cipher);
352 for(std::size_t i = 0 ; i < nbBlocks ; ++i) {
353 output << out[i];
354 }
355 index++;
356 }
357 output.flush();

360 template<typename Derived, typename Cipher>
361 inline void
362 BlockCipherModeOfOperation<Derived,Cipher>::cipherInParallel(std::istream& input,
363 std::ostream& output,
364 const std::string& key) {
365 std::array<uint8_t,Cipher::getBlockSize()>* data = new std::array<uint8_t,Cipher::getBlockSize()>[_nbThread];
366 std::array<uint8_t,Cipher::getBlockSize()>* result = new std::array<uint8_t,Cipher::getBlockSize()>[_nbThread];
367 std::size_t* nbWrite = new std::size_t[_nbThread];
368 uint32_t index = 0;
369 std::deque<std::thread> threads;
370 std::vector<Cipher> ciph;
371 ciph.push_back(Cipher(reinterpret_cast<const uint8_t*>(key.data())));
372 for(std::size_t i = 1 ; i < _nbThread ; ++i) {
373 ciph.push_back(Cipher(ciph[0]));
374 }
375 while(!input.eof()) {
376 for(std::size_t i = 0 ; i < _nbThread ; ++i) {
377 input.read(reinterpret_cast<char*>(data[i].data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
378 std::streamsize nbRead = input.gcount();
379 if(nbRead == 0) {
380 break;
381 }
382 threads.push_back(std::thread(&Derived::deferredCipherBlock,
383 this,
384 index++,
385 std::ref(data[i]),
386 nbRead,
387 std::ref(result[i]),
388 std::ref(ciph[i]),
389 std::ref(nbWrite[i])));
390 if(static_cast<std::size_t>(nbRead) != Cipher::getBlockSize()) {
391 break;
392 }
393 }
394 for(uint32_t i = 0 ; !threads.empty() ; ++i) {
395 std::thread& th = threads.front();
396 th.join();
397 for(uint32_t j = 0 ; j < nbWrite[i] ; ++j) {
398 output << result[i][j];
399 }
400 threads.pop_front();
401 }
402 }
403
404 output.flush();
405 delete[] data;
406 delete[] result;
407 delete[] nbWrite;
408 }

412 BlockCipherModeOfOperation<Derived,Cipher>::decipherSequentially(std::istream& input,
413 std::ostream& output,
414 const std::string& key,
415 std::array<uint8_t,Cipher::getBlockSize()>& prevData) {
416 Cipher cipher(reinterpret_cast<const uint8_t*>(key.data()));
417 std::array<uint8_t,Cipher::getBlockSize()> out;
418 std::array<uint8_t,Cipher::getBlockSize()> data;
419 std::array<uint8_t,Cipher::getBlockSize()> cipherData;
420 input.read(reinterpret_cast<char*>(data.data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
421 std::streamsize nbRead = input.gcount();
422 uint32_t index = 0;
423 do {
424 cipherData = data;
425 std::streamsize read = nbRead;
426 input.read(reinterpret_cast<char*>(data.data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
427
428 std::size_t end = decipherBlock(cipherData, prevData, read, input.eof(), out, index, cipher);
429 for(std::size_t i = 0 ; i < end ; ++i) {
430 output << out[i];
431 }
432
433 prevData = cipherData;
434 nbRead = input.gcount();
435 index++;
436
437 } while(nbRead != 0);
438 output.flush();
439 }

441 template<typename Derived, typename Cipher>
442 inline void
443 BlockCipherModeOfOperation<Derived,Cipher>::decipherInParallel(std::istream& input,
444 std::ostream& output,
445 const std::string& key,
446 std::array<uint8_t,Cipher::getBlockSize()>& prevData) {
447 std::array<uint8_t,Cipher::getBlockSize()>* data = new std::array<uint8_t,Cipher::getBlockSize()>[_nbThread + 1];
448 std::array<uint8_t,Cipher::getBlockSize()>* result = new std::array<uint8_t,Cipher::getBlockSize()>[_nbThread];
449 std::size_t* nbWrite = new std::size_t[_nbThread];
450 uint32_t index = 0;
451 std::deque<std::thread> threads;
452 std::vector<Cipher> ciph;
453 ciph.push_back(Cipher(reinterpret_cast<const uint8_t*>(key.data())));
454 for(std::size_t i = 1 ; i < _nbThread ; ++i) {
455 ciph.push_back(Cipher(ciph[0]));
456 }
457 bool lastBlock = false;
458 input.read(reinterpret_cast<char*>(data[0].data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
459 std::size_t nbRead = static_cast<std::size_t>(input.gcount());
460 do {
461 for(std::size_t i = 0, idx = 1 ; i < _nbThread ; ++i, ++idx) {
462 input.read(reinterpret_cast<char*>(data[idx].data()), static_cast<std::streamsize>(Cipher::getBlockSize()));
463 std::streamsize nextCount = input.gcount();
464 if(nextCount == 0) {
465 lastBlock = true;
466 }
467 threads.push_back(std::thread(&Derived::deferredDecipherBlock,
468 this,
469 index++,
470 std::ref(data[i]),
471 prevData,
472 nbRead,
473 lastBlock,
474 std::ref(result[i]),
475 std::ref(ciph[i]),
476 std::ref(nbWrite[i])));
477 if(lastBlock) {
478 break;
479 }
480 nbRead = static_cast<std::size_t>(nextCount);
481 prevData = data[i];
482 }
483 for(uint32_t i = 0 ; !threads.empty() ; ++i) {
484 std::thread& th = threads.front();
485 th.join();
486 for(uint32_t j = 0 ; j < nbWrite[i] ; ++j) {
487 output << result[i][j];
488 }
489 threads.pop_front();
490 }
491 if(_error) {
492 throw InvalidBlockException("Error while decipher stream");
493 }
494 data[0] = data[_nbThread];
495 } while(!lastBlock);
496
497 output.flush();
498 delete[] data;
499 delete[] result;
500 delete[] nbWrite;
501 }
