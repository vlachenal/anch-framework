627 ClausesBuilder::checkAndAddClause(const std::string& boolAgg, const std::string& column, ClauseMaker clause, T& value, ValueChecker<T> checker) {
628 auto optVal = checker(value);
629 if(optVal) {
630 addBooleanAggregator(boolAgg);
631 _buffer << clause(column);
632 _values.push_back(optVal.value());
633 // if(value instanceof Optional<?>) {
634 // // Optional value has already been check at his point
635 // values.add(((Optional<?>)value).get());
636 // } else if(value instanceof Collection) { // For (NOT) IN operators
637 // buffer.append(SQL.toSQLList((Collection<?>)value));
638 // } else if(value instanceof SelectBuilder) { // For (NOT) EXISTS operators
639 // buffer.append('(').append(value).append(')');
640 // values.addAll(((SelectBuilder)value).values);
641 // } else if(value instanceof SQLQuery) { // For (NOT) EXISTS operators
642 // final SQLQuery query = (SQLQuery)value;
643 // buffer.append('(').append(query.getQuery()).append(')');
644 // values.addAll(query.getValues());
645 // } else {
646 // values.add(value);
647 // }
648 _firstClause = false;
649 }
650 return *this;
651 }

655 ClausesBuilder::checkAndAddClause(const std::string& boolAgg, const std::string& column, ClauseMaker clause, T& value1, T& value2, ValueChecker<T> checker) {
656 auto optVal1 = checker(value1);
657 auto optVal2 = checker(value2);
658 if(optVal1 && optVal2) {
659 addBooleanAggregator(boolAgg);
660 _buffer << clause(column);
661 _values.push_back(optVal1.value());
662 _values.push_back(optVal2.value());
663 // if(value1 instanceof Optional<?>) {
664 // // Optional value has already been check at his point
665 // values.add(((Optional<?>)value1).get());
666 // } else {
667 // values.add(value1);
668 // }
669 // if(value2 instanceof Optional<?>) {
670 // // Optional value has already been check at his point
671 // values.add(((Optional<?>)value2).get());
672 // } else {
673 // values.add(value2);
674 // }
675 _firstClause = false;
676 }
677 return *this;
